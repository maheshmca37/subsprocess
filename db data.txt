create table tsp_schools
(
id bigint generated always as identity primary key,

school_name text not null,

created_at timestamp default now()
);


create table tsp_academic_years
(
id bigint generated always as identity primary key,

school_id bigint references tsp_schools(id),

year_name text,

is_active boolean default true
);


create table tsp_teachers
(
id bigint generated always as identity primary key,

school_id bigint references tsp_schools(id),

academic_year_id bigint references tsp_academic_years(id),

teacher_name text
);



create table tsp_classes
(
id bigint generated always as identity primary key,

school_id bigint references tsp_schools(id),

academic_year_id bigint references tsp_academic_years(id),

class_name text
);




create table tsp_timetables
(
id bigint generated always as identity primary key,

school_id bigint references tsp_schools(id),

academic_year_id bigint references tsp_academic_years(id),

teacher_id bigint references tsp_teachers(id),

class_id bigint references tsp_classes(id),

subject text,

day text,

period_no int
);

create table tsp_substitutions
(
id bigint generated always as identity primary key,

school_id bigint references tsp_schools(id),

academic_year_id bigint references tsp_academic_years(id),

substitution_date date,

day text,

period_no int,

class_id bigint references tsp_classes(id),

subject text,

absent_teacher_id bigint references tsp_teachers(id),

assigned_teacher_id bigint references tsp_teachers(id),

created_at timestamp default now(),

constraint unique_tsp_substitution

unique
(
school_id,
substitution_date,
class_id,
period_no
)

);
